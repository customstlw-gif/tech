このページについて

Snowflakeの設計・構築・運用で頻繁に使うSQLコマンドを、利用シーン別に整理したリファレンスです。各コードはそのまま実行できる形で記載しています。右上の「コピー」ボタンでクリップボードに取得できます。

SAMPLE_DB / SAMPLE_WH / SAMPLE_ROLE などの SAMPLE_ で始まる名前と '<...>' 部分は環境に合わせて置き換えてください。

ACCOUNT_USAGE… 保持1年・反映遅延あり（最大45分〜3時間程度） INFORMATION_SCHEMA… ほぼリアルタイム・保持7〜14日 要管理者権限
該当するコマンドが見つかりませんでした。

01構築時のコマンド

ロール階層 → ウェアハウス → データベース/スキーマ → 権限 → ユーザーの順で作成するのが基本です。オブジェクトの所有ロールを統一（例: SYSADMIN配下のカスタムロール）しておくと、後の権限管理が簡単になります。

1-1. ロール・ロール階層

機能ロール／アクセスロールの作成と階層化
USERADMIN / SECURITYADMIN
カスタムロールは最終的に SYSADMIN の配下に付けるのが推奨です（孤立ロールを作らない）。
USE ROLE USERADMIN;
CREATE ROLE IF NOT EXISTS SAMPLE_DEVELOPER COMMENT = '開発者用 機能ロール';
CREATE ROLE IF NOT EXISTS SAMPLE_DB_RW     COMMENT = 'SAMPLE_DB 読み書き アクセスロール';
CREATE ROLE IF NOT EXISTS SAMPLE_DB_RO     COMMENT = 'SAMPLE_DB 参照専用 アクセスロール';

USE ROLE SECURITYADMIN;
GRANT ROLE SAMPLE_DB_RO     TO ROLE SAMPLE_DB_RW;     -- RW は RO を包含
GRANT ROLE SAMPLE_DB_RW     TO ROLE SAMPLE_DEVELOPER;
GRANT ROLE SAMPLE_DEVELOPER TO ROLE SYSADMIN;         -- 階層の頂点を SYSADMIN へ
データベースロールの作成
DB単位の権限管理
DB内に閉じたロール。DBのクローンやデータ共有と一緒に権限を持ち運べます。
USE ROLE SYSADMIN;
CREATE DATABASE ROLE IF NOT EXISTS SAMPLE_DB.DR_READ;
GRANT USAGE  ON SCHEMA SAMPLE_DB.SAMPLE_SCHEMA TO DATABASE ROLE SAMPLE_DB.DR_READ;
GRANT SELECT ON ALL TABLES IN SCHEMA SAMPLE_DB.SAMPLE_SCHEMA TO DATABASE ROLE SAMPLE_DB.DR_READ;
GRANT DATABASE ROLE SAMPLE_DB.DR_READ TO ROLE SAMPLE_DB_RO;

1-2. ウェアハウス

ウェアハウスの作成（標準設定）
SYSADMIN
自動停止は60秒前後、初期状態は停止で作成するとコストのムダを防げます。用途（ETL / BI / 開発）ごとに分けるのが基本です。
USE ROLE SYSADMIN;
CREATE WAREHOUSE IF NOT EXISTS SAMPLE_WH
  WAREHOUSE_SIZE      = 'XSMALL'
  AUTO_SUSPEND        = 60          -- 秒
  AUTO_RESUME         = TRUE
  INITIALLY_SUSPENDED = TRUE
  STATEMENT_TIMEOUT_IN_SECONDS        = 3600  -- 暴走クエリ対策
  STATEMENT_QUEUED_TIMEOUT_IN_SECONDS = 600
  COMMENT = 'BI参照用';

GRANT USAGE, OPERATE ON WAREHOUSE SAMPLE_WH TO ROLE SAMPLE_DEVELOPER;
マルチクラスタ ウェアハウス（同時実行対策）
Enterprise 以上
CREATE WAREHOUSE IF NOT EXISTS SAMPLE_BI_WH
  WAREHOUSE_SIZE    = 'SMALL'
  MIN_CLUSTER_COUNT = 1
  MAX_CLUSTER_COUNT = 3
  SCALING_POLICY    = 'STANDARD'   -- コスト優先なら 'ECONOMY'
  AUTO_SUSPEND = 60  AUTO_RESUME = TRUE  INITIALLY_SUSPENDED = TRUE;
リソースモニター（クレジット上限）
ACCOUNTADMIN
USE ROLE ACCOUNTADMIN;
CREATE OR REPLACE RESOURCE MONITOR SAMPLE_RM
  WITH CREDIT_QUOTA = 100
       FREQUENCY = MONTHLY
       START_TIMESTAMP = IMMEDIATELY
  TRIGGERS ON 80  PERCENT DO NOTIFY
           ON 100 PERCENT DO SUSPEND
           ON 110 PERCENT DO SUSPEND_IMMEDIATE;

ALTER WAREHOUSE SAMPLE_WH SET RESOURCE_MONITOR = SAMPLE_RM;

1-3. データベース・スキーマ・テーブル

データベース／スキーマの作成
SYSADMIN
管理アクセススキーマ（WITH MANAGED ACCESS）にすると、権限付与をスキーマ所有者に集約でき、オブジェクト所有者による勝手なGRANTを防げます。
USE ROLE SYSADMIN;
CREATE DATABASE IF NOT EXISTS SAMPLE_DB
  DATA_RETENTION_TIME_IN_DAYS = 1      -- Time Travel 日数（Enterprise は最大90）
  COMMENT = 'サンプルDB';

CREATE SCHEMA IF NOT EXISTS SAMPLE_DB.SAMPLE_SCHEMA
  WITH MANAGED ACCESS
  COMMENT = '業務データ';

-- 一時データ用（Fail-safe なし → ストレージ費用削減）
CREATE TRANSIENT SCHEMA IF NOT EXISTS SAMPLE_DB.WORK;
テーブルの作成（クラスタリングキー・コメント付き）
DDL
CREATE TABLE IF NOT EXISTS SAMPLE_DB.SAMPLE_SCHEMA.ORDERS (
  ORDER_ID     NUMBER(38,0)  NOT NULL COMMENT '注文ID',
  CUSTOMER_ID  NUMBER(38,0)           COMMENT '顧客ID',
  ORDER_DATE   DATE                   COMMENT '注文日',
  AMOUNT       NUMBER(18,2)           COMMENT '金額',
  CREATED_AT   TIMESTAMP_NTZ DEFAULT CURRENT_TIMESTAMP(),
  CONSTRAINT PK_ORDERS PRIMARY KEY (ORDER_ID)   -- 情報制約（強制はされない）
)
CLUSTER BY (ORDER_DATE)   -- 数TB規模以上で検討
COMMENT = '注文テーブル';

1-4. 権限付与（将来付与を含む）

アクセスロールへの権限付与テンプレート
SECURITYADMIN
FUTURE GRANTS を併用すると、今後作成されるテーブル・ビューにも自動で権限が付きます。
USE ROLE SECURITYADMIN;
-- 参照専用（RO）
GRANT USAGE  ON DATABASE SAMPLE_DB                       TO ROLE SAMPLE_DB_RO;
GRANT USAGE  ON ALL SCHEMAS    IN DATABASE SAMPLE_DB     TO ROLE SAMPLE_DB_RO;
GRANT USAGE  ON FUTURE SCHEMAS IN DATABASE SAMPLE_DB     TO ROLE SAMPLE_DB_RO;
GRANT SELECT ON ALL TABLES     IN DATABASE SAMPLE_DB     TO ROLE SAMPLE_DB_RO;
GRANT SELECT ON FUTURE TABLES  IN DATABASE SAMPLE_DB     TO ROLE SAMPLE_DB_RO;
GRANT SELECT ON ALL VIEWS      IN DATABASE SAMPLE_DB     TO ROLE SAMPLE_DB_RO;
GRANT SELECT ON FUTURE VIEWS   IN DATABASE SAMPLE_DB     TO ROLE SAMPLE_DB_RO;

-- 読み書き（RW）
GRANT INSERT, UPDATE, DELETE, TRUNCATE ON ALL TABLES    IN DATABASE SAMPLE_DB TO ROLE SAMPLE_DB_RW;
GRANT INSERT, UPDATE, DELETE, TRUNCATE ON FUTURE TABLES IN DATABASE SAMPLE_DB TO ROLE SAMPLE_DB_RW;
GRANT CREATE TABLE, CREATE VIEW ON SCHEMA SAMPLE_DB.SAMPLE_SCHEMA TO ROLE SAMPLE_DB_RW;
注意：スキーマ単位とデータベース単位の FUTURE GRANTS が両方ある場合、スキーマ単位が優先され、データベース単位は適用されません。どちらかに統一してください。

1-5. ユーザー

人ユーザー／サービスユーザーの作成
USERADMIN
バッチやツール接続用は TYPE = SERVICE（パスワード不可）＋キーペア認証にします。
USE ROLE USERADMIN;
-- 人ユーザー（SSO 利用時はパスワード不要）
CREATE USER IF NOT EXISTS TARO_YAMADA
  LOGIN_NAME = 'taro.yamada@example.com'
  EMAIL = 'taro.yamada@example.com'
  DEFAULT_ROLE = SAMPLE_DEVELOPER
  DEFAULT_WAREHOUSE = SAMPLE_WH
  TYPE = PERSON;

-- サービスユーザー（キーペア認証）
CREATE USER IF NOT EXISTS SVC_ETL
  TYPE = SERVICE
  RSA_PUBLIC_KEY = '<公開鍵（ヘッダ/フッタ除く）>'
  DEFAULT_ROLE = SAMPLE_DB_RW
  DEFAULT_WAREHOUSE = SAMPLE_WH;

USE ROLE SECURITYADMIN;
GRANT ROLE SAMPLE_DEVELOPER TO USER TARO_YAMADA;
GRANT ROLE SAMPLE_DB_RW     TO USER SVC_ETL;

1-7. データロード基盤（外部連携）

ストレージ統合（S3の例）＋外部ステージ
ACCOUNTADMIN
作成後に DESC INTEGRATION で表示される IAMユーザーARN と External ID を AWS 側の信頼ポリシーへ設定します。
USE ROLE ACCOUNTADMIN;
CREATE STORAGE INTEGRATION IF NOT EXISTS SAMPLE_S3_INT
  TYPE = EXTERNAL_STAGE
  STORAGE_PROVIDER = 'S3'
  ENABLED = TRUE
  STORAGE_AWS_ROLE_ARN = 'arn:aws:iam::123456789012:role/snowflake-access'
  STORAGE_ALLOWED_LOCATIONS = ('s3://sample-bucket/landing/');

DESC INTEGRATION SAMPLE_S3_INT;   -- STORAGE_AWS_IAM_USER_ARN / STORAGE_AWS_EXTERNAL_ID を確認
GRANT USAGE ON INTEGRATION SAMPLE_S3_INT TO ROLE SYSADMIN;

USE ROLE SYSADMIN;
CREATE FILE FORMAT IF NOT EXISTS SAMPLE_DB.SAMPLE_SCHEMA.FF_CSV
  TYPE = CSV  SKIP_HEADER = 1  FIELD_OPTIONALLY_ENCLOSED_BY = '"'
  NULL_IF = ('', 'NULL')  ENCODING = 'UTF8';

CREATE STAGE IF NOT EXISTS SAMPLE_DB.SAMPLE_SCHEMA.STG_LANDING
  URL = 's3://sample-bucket/landing/'
  STORAGE_INTEGRATION = SAMPLE_S3_INT
  FILE_FORMAT = SAMPLE_DB.SAMPLE_SCHEMA.FF_CSV;

LIST @SAMPLE_DB.SAMPLE_SCHEMA.STG_LANDING;   -- 疎通確認
COPY / Snowpipe / ストリーム / タスク
ELT パイプライン
-- 一括ロード
COPY INTO SAMPLE_DB.SAMPLE_SCHEMA.ORDERS
  FROM @SAMPLE_DB.SAMPLE_SCHEMA.STG_LANDING/orders/
  ON_ERROR = 'ABORT_STATEMENT';

-- 継続ロード（Snowpipe 自動取り込み）
CREATE PIPE IF NOT EXISTS SAMPLE_DB.SAMPLE_SCHEMA.PIPE_ORDERS
  AUTO_INGEST = TRUE
AS
  COPY INTO SAMPLE_DB.SAMPLE_SCHEMA.ORDERS
  FROM @SAMPLE_DB.SAMPLE_SCHEMA.STG_LANDING/orders/;

-- 変更データ取得
CREATE STREAM IF NOT EXISTS SAMPLE_DB.SAMPLE_SCHEMA.ORDERS_STREAM
  ON TABLE SAMPLE_DB.SAMPLE_SCHEMA.ORDERS;

-- 定期実行タスク（ストリームにデータがある時のみ実行）
CREATE TASK IF NOT EXISTS SAMPLE_DB.SAMPLE_SCHEMA.TASK_MERGE_ORDERS
  WAREHOUSE = SAMPLE_WH
  SCHEDULE  = 'USING CRON 0 * * * * Asia/Tokyo'
  WHEN SYSTEM$STREAM_HAS_DATA('SAMPLE_DB.SAMPLE_SCHEMA.ORDERS_STREAM')
AS
  INSERT INTO SAMPLE_DB.SAMPLE_SCHEMA.ORDERS_HIST
  SELECT * EXCLUDE (METADATA$ACTION, METADATA$ISUPDATE, METADATA$ROW_ID)
  FROM SAMPLE_DB.SAMPLE_SCHEMA.ORDERS_STREAM;

ALTER TASK SAMPLE_DB.SAMPLE_SCHEMA.TASK_MERGE_ORDERS RESUME;   -- 作成直後は SUSPENDED
ダイナミックテーブル
宣言的パイプライン
CREATE OR REPLACE DYNAMIC TABLE SAMPLE_DB.SAMPLE_SCHEMA.DT_DAILY_SALES
  TARGET_LAG = '1 hour'
  WAREHOUSE  = SAMPLE_WH
AS
  SELECT ORDER_DATE, SUM(AMOUNT) AS TOTAL_AMOUNT
  FROM SAMPLE_DB.SAMPLE_SCHEMA.ORDERS
  GROUP BY ORDER_DATE;

03性能情報の取得

遅いクエリの特定 → ボトルネック（キュー待ち・スピル・プルーニング不足）の切り分け → ウェアハウスの負荷確認、の順で調べます。

3-1. クエリ実行状況

実行時間の長いクエリ TOP 20（過去7日）
ACCOUNT_USAGE
SELECT QUERY_ID, USER_NAME, ROLE_NAME, WAREHOUSE_NAME, WAREHOUSE_SIZE,
       TOTAL_ELAPSED_TIME/1000      AS ELAPSED_SEC,
       COMPILATION_TIME/1000        AS COMPILE_SEC,
       EXECUTION_TIME/1000          AS EXEC_SEC,
       QUEUED_OVERLOAD_TIME/1000    AS QUEUED_SEC,
       BYTES_SCANNED/POWER(1024,3)  AS SCANNED_GB,
       LEFT(QUERY_TEXT, 200)        AS QUERY_TEXT
FROM SNOWFLAKE.ACCOUNT_USAGE.QUERY_HISTORY
WHERE START_TIME >= DATEADD(DAY, -7, CURRENT_TIMESTAMP())
  AND EXECUTION_STATUS = 'SUCCESS'
ORDER BY TOTAL_ELAPSED_TIME DESC
LIMIT 20;
直近・実行中のクエリ（リアルタイム）
INFORMATION_SCHEMA
SELECT QUERY_ID, USER_NAME, WAREHOUSE_NAME, EXECUTION_STATUS,
       START_TIME, TOTAL_ELAPSED_TIME/1000 AS ELAPSED_SEC,
       LEFT(QUERY_TEXT, 200) AS QUERY_TEXT
FROM TABLE(INFORMATION_SCHEMA.QUERY_HISTORY(
       END_TIME_RANGE_START => DATEADD(HOUR, -1, CURRENT_TIMESTAMP()),
       RESULT_LIMIT => 1000))
WHERE EXECUTION_STATUS IN ('RUNNING', 'QUEUED', 'BLOCKED')
ORDER BY START_TIME;

-- 特定ウェアハウス / ユーザーに絞る場合
-- TABLE(INFORMATION_SCHEMA.QUERY_HISTORY_BY_WAREHOUSE(WAREHOUSE_NAME => 'SAMPLE_WH'))
-- TABLE(INFORMATION_SCHEMA.QUERY_HISTORY_BY_USER(USER_NAME => 'SVC_ETL'))
同じ処理の傾向を見る（パラメータ化ハッシュで集計）
ACCOUNT_USAGE
リテラル値違いの同型クエリをまとめて、合計負荷の大きい処理を特定します。
SELECT QUERY_PARAMETERIZED_HASH,
       COUNT(*)                               AS EXEC_COUNT,
       ROUND(SUM(TOTAL_ELAPSED_TIME)/1000/60) AS TOTAL_MIN,
       ROUND(AVG(TOTAL_ELAPSED_TIME)/1000, 1) AS AVG_SEC,
       ANY_VALUE(LEFT(QUERY_TEXT, 200))       AS SAMPLE_QUERY
FROM SNOWFLAKE.ACCOUNT_USAGE.QUERY_HISTORY
WHERE START_TIME >= DATEADD(DAY, -7, CURRENT_TIMESTAMP())
GROUP BY 1
ORDER BY TOTAL_MIN DESC
LIMIT 20;

3-2. ボトルネックの切り分け

ディスクスピルが発生したクエリ（メモリ不足）
ACCOUNT_USAGE
リモートスピルが多い場合は、ウェアハウスのサイズアップかクエリ改善（結合・集計の見直し）を検討します。
SELECT QUERY_ID, WAREHOUSE_NAME, WAREHOUSE_SIZE,
       ROUND(BYTES_SPILLED_TO_LOCAL_STORAGE /POWER(1024,3), 2) AS LOCAL_SPILL_GB,
       ROUND(BYTES_SPILLED_TO_REMOTE_STORAGE/POWER(1024,3), 2) AS REMOTE_SPILL_GB,
       TOTAL_ELAPSED_TIME/1000 AS ELAPSED_SEC,
       LEFT(QUERY_TEXT, 200) AS QUERY_TEXT
FROM SNOWFLAKE.ACCOUNT_USAGE.QUERY_HISTORY
WHERE START_TIME >= DATEADD(DAY, -7, CURRENT_TIMESTAMP())
  AND (BYTES_SPILLED_TO_LOCAL_STORAGE > 0 OR BYTES_SPILLED_TO_REMOTE_STORAGE > 0)
ORDER BY BYTES_SPILLED_TO_REMOTE_STORAGE DESC, BYTES_SPILLED_TO_LOCAL_STORAGE DESC
LIMIT 20;
プルーニング効率の悪いクエリ（全件スキャンに近い）
ACCOUNT_USAGE
SELECT QUERY_ID, WAREHOUSE_NAME, PARTITIONS_SCANNED, PARTITIONS_TOTAL,
       ROUND(PARTITIONS_SCANNED / NULLIF(PARTITIONS_TOTAL,0) * 100, 1) AS SCAN_RATIO_PCT,
       TOTAL_ELAPSED_TIME/1000 AS ELAPSED_SEC,
       LEFT(QUERY_TEXT, 200) AS QUERY_TEXT
FROM SNOWFLAKE.ACCOUNT_USAGE.QUERY_HISTORY
WHERE START_TIME >= DATEADD(DAY, -7, CURRENT_TIMESTAMP())
  AND PARTITIONS_TOTAL > 1000
  AND PARTITIONS_SCANNED / NULLIF(PARTITIONS_TOTAL,0) > 0.8
ORDER BY TOTAL_ELAPSED_TIME DESC
LIMIT 20;

WITH accessed AS (
  SELECT DISTINCT obj.value:"objectName"::STRING AS FQN
  FROM SNOWFLAKE.ACCOUNT_USAGE.ACCESS_HISTORY ah,
       LATERAL FLATTEN(INPUT => ah.BASE_OBJECTS_ACCESSED) obj
  WHERE ah.QUERY_START_TIME >= DATEADD(DAY, -90, CURRENT_TIMESTAMP())
)
SELECT t.TABLE_CATALOG||'.'||t.TABLE_SCHEMA||'.'||t.TABLE_NAME AS FQN,
       ROUND(t.BYTES/POWER(1024,3), 2) AS SIZE_GB, t.LAST_ALTERED
FROM SNOWFLAKE.ACCOUNT_USAGE.TABLES t
LEFT JOIN accessed a
  ON a.FQN = t.TABLE_CATALOG||'.'||t.TABLE_SCHEMA||'.'||t.TABLE_NAME
WHERE t.DELETED IS NULL AND t.TABLE_TYPE = 'BASE TABLE' AND a.FQN IS NULL
ORDER BY SIZE_GB DESC NULLS LAST;

  AT(TIMESTAMP => '2026-11-14 09:00:00 +0900'::TIMESTAMP_TZ);

CREATE OR REPLACE ALERT SAMPLE_DB.MONITOR.ALERT_TASK_FAILED
  WAREHOUSE = SAMPLE_WH
  SCHEDULE  = '10 MINUTE'
  IF (EXISTS (
        SELECT 1
        FROM TABLE(SAMPLE_DB.INFORMATION_SCHEMA.TASK_HISTORY(
               SCHEDULED_TIME_RANGE_START => SNOWFLAKE.ALERT.LAST_SUCCESSFUL_SCHEDULED_TIME(),
               SCHEDULED_TIME_RANGE_END   => SNOWFLAKE.ALERT.SCHEDULED_TIME()))
        WHERE STATE = 'FAILED'))
  THEN
    CALL SYSTEM$SEND_EMAIL('SAMPLE_EMAIL_INT', 'ops-team@example.com',
                           '[Snowflake] タスク失敗検知',
                           'TASK_HISTORY を確認してください。');
